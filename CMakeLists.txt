cmake_minimum_required(VERSION 3.22)
project(INXOMNIA VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "Minimum macOS version")
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(INXOMNIA COMPANY_NAME "Dreamdaw" PLUGIN_MANUFACTURER_CODE Drmd PLUGIN_CODE DmL1 FORMATS VST3 PRODUCT_NAME "INXOMNIA" VST3_CATEGORIES Fx\;Mastering IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(INXOMNIA PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/DspEngine.cpp)
target_compile_definitions(INXOMNIA PUBLIC JUCE_WEB_BROWSER=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0)
target_link_libraries(INXOMNIA PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
if(UNIX AND NOT APPLE)
 find_package(CURL REQUIRED)
 target_link_libraries(INXOMNIA PRIVATE CURL::libcurl)
endif()
if(WIN32)
 target_link_libraries(INXOMNIA PRIVATE Advapi32)
endif()

if(BUILD_TESTING)
 add_executable(DreamMasterDSPTests Tests/DspEngineTests.cpp Source/DspEngine.cpp)
 target_include_directories(DreamMasterDSPTests PRIVATE Source)
 target_link_libraries(DreamMasterDSPTests PRIVATE juce::juce_data_structures)
 if(UNIX AND NOT APPLE)
  target_link_libraries(DreamMasterDSPTests PRIVATE CURL::libcurl)
 endif()
 add_test(NAME DreamMasterDSPTests COMMAND DreamMasterDSPTests)
endif()
